reorderRules(ruleIds: string[]): CategorizationRuleSummary[] {
  const updateOrder = this.sqlite.transaction(() => {
    ruleIds.forEach((id, index) => {
      this.sqlite.prepare(
        'UPDATE categorization_rules SET sort_order = ?, updated_at = ? WHERE id = ? AND is_system = 0'
      ).run(index + 1, nowIso(), id)
    })
  })
  updateOrder()
  return this.listRules()
}

2. Add `applyAllRulesToTransactions(batchId: string)` method (per Research Pattern 6 and D-09, D-12):
```typescript
applyAllRulesToTransactions(batchId: string): Array<{ ruleName: string; count: number }> {
  // Fetch enabled user rules in sort_order ASC (D-09: enabled user rules only)
  const rules = this.sqlite.prepare(
    `SELECT id, name, condition_json, action_json FROM categorization_rules
     WHERE is_enabled = 1 AND is_system = 0
     ORDER BY sort_order ASC`
  ).all() as Array<{ id: string; name: string; condition_json: string; action_json: string }>

  // Fetch uncategorized transactions from this batch (Pitfall 7: only category_id IS NULL)
  const transactions = this.sqlite.prepare(
    `SELECT id, description, signed_amount_minor, normalized_type, tags_json, direction
     FROM imported_transactions
     WHERE import_batch_id = ? AND (category_id IS NULL OR category_id = '')`
  ).all(batchId) as Array<any>

  const counts = new Map<string, number>()

  const applyTransaction = this.sqlite.transaction(() => {
    for (const tx of transactions) {
      for (const rule of rules) {
        const condition = JSON.parse(rule.condition_json)
        const action = JSON.parse(rule.action_json)
        if (this.matchesRuleCondition(tx, condition)) {
          // Apply the rule action (reuse existing applyRuleActionToTransaction pattern)
          if (action.categoryId) {
            const categoryPath = this.getCategoryPathById(action.categoryId)
            this.sqlite.prepare(
              'UPDATE imported_transactions SET category_id = ?, category_label = ?, updated_at = ? WHERE id = ?'
            ).run(action.categoryId, categoryPath.join(' > '), nowIso(), tx.id)
          }
          if (action.appendTags?.length) {
            const existingTags = JSON.parse(tx.tags_json || '[]')
            const merged = [...new Set([...existingTags, ...action.appendTags])]
            this.sqlite.prepare(
              'UPDATE imported_transactions SET tags_json = ?, updated_at = ? WHERE id = ?'
            ).run(JSON.stringify(merged), nowIso(), tx.id)
          }

          // Emit audit event (D-12)
          this.insertAuditEvent({
            eventType: 'transaction:auto-categorized',
            metadata: {
              transactionId: tx.id,
              ruleName: rule.name,
              ruleId: rule.id,
              categoryId: action.categoryId || null,
              batchId
            }
          })

          counts.set(rule.name, (counts.get(rule.name) || 0) + 1)
          break // first match wins (D-05)
        }
      }
    }
  })
  applyTransaction()

  return Array.from(counts.entries())
    .map(([ruleName, count]) => ({ ruleName, count }))
    .filter(entry => entry.count > 0)
}
```
