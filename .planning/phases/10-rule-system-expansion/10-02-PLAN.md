1. In `src/main/persistence/db.ts`, rewrite `matchesRuleCondition()`:

Replace the existing descriptionContains matching block with:
```typescript
// descriptionTerms: AND across all entries
if (condition.descriptionTerms && condition.descriptionTerms.length > 0) {
  const text = row.description.toLowerCase()
  const allMatch = condition.descriptionTerms.every((term) => {
    const v = term.value.toLowerCase()
    switch (term.op) {
      case 'contains':    return text.includes(v)
      case 'starts-with': return text.startsWith(v)
      case 'ends-with':   return text.endsWith(v)
      case 'regex': {
        try { return new RegExp(term.value, 'i').test(row.description) }
        catch { return false }
      }
      default: return false
    }
  })
  if (!allMatch) return false
}
```
Keep the existing `amountMinMinor`, `amountMaxMinor`, `transactionTypes`, `tags`, and `directions` checks unchanged — they are ANDed with the descriptionTerms result.

2. Add a renderer-side ReDoS detection utility. Create the function in `RuleEditorPanel.tsx` (or a small utility if preferred):
```typescript
const REDOS_RISK_PATTERNS = [
  /\([^)]*[+*][^)]*\)[+*]/,   // (x+)+ nested quantifiers
  /\([^)]*\|[^)]*\)[+*]/,     // (a|b)+ alternation + quantifier
  /\.[+*][+*]/,                // .++ repetition
]

export const detectReDoSRisk = (pattern: string): boolean =>
  REDOS_RISK_PATTERNS.some((check) => check.test(pattern))
```
