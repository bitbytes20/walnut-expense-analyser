    ```typescript
    export const changePin = async (currentPin: string, newPin: string): Promise<{ ok: boolean; error?: string }> => {
      const repository = getWalnutRepository()
      const state = repository.loadAppState().security

      if (!state.pinHash) {
        return { ok: false, error: 'No PIN is set.' }
      }

      const matched = await verify(state.pinHash, currentPin)
      if (!matched) {
        repository.logSecurityEvent('security.pin_change_failed', { reason: 'wrong_current_pin' })
        return { ok: false, error: 'Current PIN is incorrect.' }
      }

      if (!isValidPin(newPin)) {
        return { ok: false, error: 'New PIN must be numeric and at least 6 digits long.' }
      }

      const newHash = await hash(newPin)
      repository.updateSecurityState({ pinHash: newHash })
      repository.logSecurityEvent('security.pin_changed', { changedAt: new Date().toISOString() })
      return { ok: true }
    }
    ```

    **Step 2 — Add clearTransactionsAndAudit to db.ts:**

    Add a public method:
    ```typescript
    clearTransactionsAndAudit(): ClearTransactionsResult {
      // Per D-13: deletes all transaction records AND associated audit events
      const txnIds = this.sqlite
        .prepare('SELECT id FROM imported_transactions')
        .all()
        .map((r: any) => r.id as string)

      const deletedCount = txnIds.length

      this.sqlite.exec('BEGIN')
      try {
        // Delete audit events that reference transaction entity_ids
        if (txnIds.length > 0) {
          // Batch delete in chunks to avoid SQLite variable limit
          for (let i = 0; i < txnIds.length; i += 500) {
            const chunk = txnIds.slice(i, i + 500)
            const placeholders = chunk.map(() => '?').join(',')
            this.sqlite.prepare(`DELETE FROM audit_events WHERE entity_id IN (${placeholders})`).run(...chunk)
          }
        }
        // Also delete transaction-category audit event types that may not have entity_id
        this.sqlite.prepare("DELETE FROM audit_events WHERE category = 'transaction'").run()

        // Delete transactions and related import records
        this.sqlite.exec(`
          DELETE FROM imported_transactions;
          DELETE FROM review_items;
          DELETE FROM import_source_files;
          DELETE FROM import_attempts;
          DELETE FROM import_batches;
        `)
        this.sqlite.exec('COMMIT')
      } catch (e) {
        this.sqlite.exec('ROLLBACK')
        throw e
      }

      return { deletedCount }
    }
    ```
